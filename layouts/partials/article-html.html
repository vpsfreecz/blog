{{/* Render legacy media and English navigation without changing article sources. */}} {{- $prefix := "/wp-content/uploads/" -}} {{- $themePrefix := "/theme/legacy-embeds/" -}} {{- if .absolute -}} {{- $base := urls.Parse .page.Site.BaseURL -}} {{- $origin := printf "%s://%s" $base.Scheme $base.Host -}} {{- $prefix = printf "%s%s" $origin $prefix -}} {{- $themePrefix = printf "%s%s" $origin $themePrefix -}} {{- end -}} {{- $images := dict "https://discourse.vpsfree.cz/uploads/default/original/1X/1c6e46d683e6dfe2c5a48bebd15f08379f514d64.png" "1c6e46d683e6dfe2c5a48bebd15f08379f514d64.png" "https://discourse.vpsfree.cz/uploads/default/optimized/1X/ff84bffeca5a0820b5637be2c6d6d760da968e0b_2_690x462.png" "ff84bffeca5a0820b5637be2c6d6d760da968e0b_2_690x462.png" "https://discourse.vpsfree.cz/images/emoji/twitter/slight_smile.png?v=12" "slight_smile.png" -}} {{- $html := .html -}} {{- range findRE `(?is)<(?:a|img|source)\b[^>]*>` $html -}} {{- $tag := replaceRE `(?:https?:)?//blog\.vpsfree\.cz/wp-content/uploads/` $prefix . -}} {{- range $source, $file := $images -}}{{- $tag = replace $tag $source (printf "%s%s" $themePrefix $file) -}}{{- end -}} {{- $html = replace $html . $tag -}} {{- end -}} {{/* Resolve English navigation offline; source links and non-navigation content stay intact. */}} {{- if eq .page.Language.Lang "en" -}} {{- $originals := dict -}} {{- range $id, $article := hugo.Data.articles -}} {{- with .discussion_url -}}{{- $originals = merge $originals (dict (urls.Parse .).Path (dict "id" $id "source" . "legacy_id" $article.legacy_post_id)) -}}{{- end -}} {{- end -}} {{- $english := dict -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang "en" -}} {{- range .RegularPages -}}{{- if and (eq .Type "posts") .Params.article_id -}} {{- $english = merge $english (dict .Params.article_id .) -}} {{- end -}}{{- end -}} {{- end -}}{{- end -}} {{- $parts := slice -}} {{- $codeDepth := 0 -}} {{- range findRE `(?is)|<(?:[^>"']|"[^"]*"|'[^']*')*>|[^<]+|<` $html -}} {{- $token := . -}} {{- if findRE `(?i)^<\s*/\s*(?:pre|code|script|style)\s*>` $token -}} {{- $codeDepth = math.Max 0 (sub $codeDepth 1) -}} {{- else if findRE `(?i)^<\s*(?:pre|code|script|style)(?:\s|>)` $token -}} {{- $codeDepth = add $codeDepth 1 -}} {{- else if and (eq $codeDepth 0) (findRE `(?i)^)` $token) -}} {{- $attributes := slice -}} {{- range findRE `(?is)^[^\s>]+|\s+[^\s=<>/]+(?:\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+))?|\s+|.` $token -}} {{- $attribute := . -}} {{- if findRE `(?is)^\s+href\s*=` $attribute -}} {{- $value := replaceRE `(?is)^\s+href\s*=\s*` "" $attribute -}} {{- $quote := "" -}} {{- if or (hasPrefix $value `"`) (hasPrefix $value `'`) -}} {{- $quote = substr $value 0 1 -}} {{- $value = strings.TrimSuffix $quote (strings.TrimPrefix $quote $value) -}} {{- end -}} {{- $target := partial "english-link.html" (dict "href" $value "originals" $originals "english" $english) -}} {{- if ne $target $value -}} {{- $lead := index (findRE `(?is)^\s+href\s*=\s*` $attribute 1) 0 -}} {{- $attribute = printf "%s%s%s%s" $lead $quote $target $quote -}} {{- end -}} {{- end -}} {{- $attributes = $attributes | append $attribute -}} {{- end -}} {{- $token = delimit $attributes "" -}} {{- end -}} {{- $parts = $parts | append $token -}} {{- end -}} {{- $html = delimit $parts "" -}} {{- end -}} {{- return ($html | safeHTML) -}}
